# lc001 无重复字符最长子串
## 码
```javascript
/**
 * @param {string} s
 * @return {number}
 */
var lengthOfLongestSubstring = function(s) {
    if(s.length <= 1){
        return s.length;
    }
    let max = 0;
    let ind_left = 0;
    let ind_right = 0;
    let used = {};
    while(ind_right < s.length - 1){
        if(ind_left === 0 && ind_right === 0){
            used[s.charAt(0)] = true;
            ind_right++;
        } else if(!used[s.charAt(ind_right)]){
            used[s.charAt(ind_right)] = true;
            ind_right++;

        } else{
            used[s.charAt(ind_left)] = false;
            ind_left++;
        }
        max = Math.max(max, (ind_right - ind_left + 1));

    }
    return max;
    
};
```
