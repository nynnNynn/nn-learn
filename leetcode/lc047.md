# lc047重复全排列
```javascript
/**
 * @param {number[]} nums
 * @return {number[][]}
 */
var permuteUnique = function(nums) {
    nums = nums.sort((a,b) => a - b);
    let result = [];
    let used = new Array(nums.length).fill(false);
    function backtrace(temp, used){
        if(temp.length === nums.length){
            result.push([...temp]);
            return;
        }
        for(let i = 0; i < nums.length; i++){
            
            if(used[i] === true){
                continue;
            }
            if(i >= 1 && nums[i] === nums[i-1] && used[i-1] === false){
                continue;
            }
            temp.push(nums[i]);
            used[i] = true;
            backtrace([...temp], used);
            temp.pop();
            used[i] = false;

        }
    }
    backtrace([], used);
    return result; 
};
```
