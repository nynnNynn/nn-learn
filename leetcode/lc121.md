# lc121 买卖股票最佳时机
```javascript
/**
 * @param {number[]} prices
 * @return {number}
 */
var maxProfit = function(prices) {
  let min = Infinity;
  let max = 0;
  for(let i = 0; i < prices.length; i++){
    min = Math.min(min, prices[i]);
    // 持续刷新最小
    max = Math.max((prices[i] - min), max);
  }
  return max;  
};
```
