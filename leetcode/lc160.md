# lc160 相交链表
```javascript
/**
 * Definition for singly-linked list.
 * function ListNode(val) {
 *     this.val = val;
 *     this.next = null;
 * }
 */

/**
 * @param {ListNode} headA
 * @param {ListNode} headB
 * @return {ListNode}
 */
var getIntersectionNode = function(headA, headB) {
    let head1 = headA;
    let head2 = headB;
    while(head1.next !== null){
        head1 = head1.next;
    }
    // 走到A最后一个了,
    head1 = headB;
    while(head1.next !== null){
        head1 = head1.next;
    }

    while(head2.next !== null){
        head2 = head2.next;
    }
    // 走到A最后一个了,
    head2 = headB;
    while(head2.next !== null){
        head2 = head2.next;
    }
    if(head1 === head2){
        return head1.val;
    }
    return false;
    
};
```
