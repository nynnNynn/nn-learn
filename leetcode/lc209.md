# lc209 长度最小子数组
```javascript
/**
 * @param {number} target
 * @param {number[]} nums
 * @return {number}
 */
var minSubArrayLen = function(target, nums) {
    let result = Infinity;
    let ind = 0;
    let temp = 0;
    let temp_num = 0;
    while(ind < nums.length){
        if((temp + nums[ind]) > target){
            temp = 0;
            temp_num = 0;
            ind++;
            continue;
        }
        if((temp + nums[ind]) = target){
            temp_num++;
            result = Math.min(result, temp_num);
            temp = 0;
            temp_num = 0;
            ind++;
            continue;
        }
        temp_num++;
        temp = temp + nums[ind];
        ind++;
    }
    return result === Infinity ? 0 : result;
};
```
