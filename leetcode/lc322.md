# lc322 零钱兑换
```javascript
/**
 * @param {number[]} coins
 * @param {number} amount
 * @return {number}
 */
var coinChange = function(coins, amount) {
    let dp = new Array(amount + 1).fill(Infinity);
    dp[0] = 0;
    for(let i = 1; i <= amount; i++){
        for(coin of coins){
            if(i < coin){
                continue;
            }
            if(i === coin){
                dp[i] = 1;
                continue;
            }
            dp[i] = Math.min(dp[i - coin] + 1, dp[i]);
        }
    }
    console.log(dp);
    return dp[amount] === Infinity? -1 : dp[amount];
};
```
